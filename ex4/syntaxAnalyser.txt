%right '='
%left '+' '-'

/* ---------- GRAMMAR RULES ---------- */

program:
    stmt_list
    ;

stmt_list:
    /* empty */
    | stmt_list stmt
    ;

stmt:
    expr ';'
    | BREAK ';'
    | switch_stmt
    | compound_stmt
    ;

compound_stmt:
    '{' stmt_list '}'
    ;

/* Switch statement grammar */
switch_stmt:
    SWITCH '(' expr ')' switch_body
        { printf(">>> Valid switch statement parsed.\n"); }
    ;

switch_body:
    '{' case_list '}'
    ;

case_list:
    /* empty */
    | case_list case_item
    ;

case_item:
    case_labels stmt_list
    ;

/* Handles single and multiple case labels (fall-through) */
case_labels:
    CASE expr ':'
    | case_labels CASE expr ':'
    | DEFAULT ':'
    | case_labels DEFAULT ':'
    ;

/* Expression grammar */
expr:
    INTEGER
    | IDENTIFIER
    | expr '=' expr
    | expr '+' expr
    | expr '-' expr
    | expr '*' expr
    | expr '/' expr
    | expr '%' expr
    | '(' expr ')'
    ;

%%
/* ---------- USER SUBROUTINES ---------- */

void yyerror(const char *s) {
    fprintf(stderr, "Syntax Error: %s\n", s);
}

int main(int argc, char *argv[]) {
    /* Read from file if provided, otherwise stdin */
    if (argc > 1) {
        yyin = fopen(argv[1], "r");
        if (!yyin) {
            perror("Error opening file");
            return 1;
        }
    }

    printf("C Switch Statement Syntax Analyzer\n");
    printf("==================================\n\n");

    yyparse();

    if (argc > 1) {
        fclose(yyin);
    }

    printf("\nParsing completed.\n");
    return 0;
}
